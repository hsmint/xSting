cmake_minimum_required(VERSION 3.16)
project(xsting_robot)
find_package(ament_cmake REQUIRED)
find_package(Threads REQUIRED)
find_package(rclcpp REQUIRED)
find_package(xsting_interfaces REQUIRED)
add_library(${PROJECT_NAME}_core src/arm/command_check.cpp)
target_compile_features(${PROJECT_NAME}_core PUBLIC cxx_std_17)
target_compile_options(${PROJECT_NAME}_core PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(${PROJECT_NAME}_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(${PROJECT_NAME}_core PUBLIC Threads::Threads)
target_link_libraries(${PROJECT_NAME}_core PUBLIC rclcpp::rclcpp xsting_interfaces::xsting_interfaces__rosidl_generator_cpp)
install(TARGETS ${PROJECT_NAME}_core EXPORT export_${PROJECT_NAME} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(Threads)
ament_export_dependencies(rclcpp xsting_interfaces)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_xsting_robot test/test_robot_contracts.cpp)
  target_link_libraries(test_xsting_robot ${PROJECT_NAME}_core)
  find_package(xsting_vision REQUIRED)
  ament_add_gtest(test_perception_to_arm test/test_perception_to_arm.cpp)
  target_link_libraries(test_perception_to_arm ${PROJECT_NAME}_core xsting_vision::xsting_vision_core)
endif()
ament_package()
